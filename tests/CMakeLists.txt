find_library(AVISYNTH_LIBRARY NAMES avisynth)
if(NOT AVISYNTH_LIBRARY)
  message(FATAL_ERROR "MEDIAN_BUILD_TESTS requires the AviSynth+ runtime library")
endif()
add_executable(median_integration integration.cpp)
set_target_properties(median_integration PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(median_integration PRIVATE ../Median)
target_link_libraries(median_integration PRIVATE ${AVISYNTH_LIBRARY})
target_compile_definitions(median_integration PRIVATE MEDIAN_TEST_HAS_OPENCL=$<BOOL:${MEDIAN_ENABLE_OPENCL}>)
add_test(NAME median_integration COMMAND median_integration $<TARGET_FILE:median>)
set_tests_properties(median_integration PROPERTIES TIMEOUT 120)
file(GLOB MEDIAN_SIMD_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../Median/simd_*.cpp")
add_executable(median_kernels kernels.cpp ../Median/kernels.cpp ${MEDIAN_SIMD_SOURCES})
include(../Median/Cpu.cmake)
median_configure_cpu(median_kernels)
target_include_directories(median_kernels PRIVATE ../Median)
add_test(NAME median_kernels COMMAND median_kernels)
add_executable(median_benchmark benchmark.cpp ../Median/kernels.cpp ${MEDIAN_SIMD_SOURCES})
target_include_directories(median_benchmark PRIVATE ../Median)
median_configure_cpu(median_benchmark)
if(MEDIAN_ENABLE_OPENCL)
  find_package(OpenCL 1.2 REQUIRED)
  add_executable(median_opencl opencl.cpp ../Median/opencl.cpp ../Median/kernels.cpp)
  target_include_directories(median_opencl PRIVATE ../Median)
  target_compile_definitions(median_opencl PRIVATE MEDIAN_ENABLE_OPENCL=1 CL_TARGET_OPENCL_VERSION=120)
  target_link_libraries(median_opencl PRIVATE OpenCL::OpenCL)
  add_test(NAME median_opencl COMMAND median_opencl)
  add_test(NAME median_opencl_integration COMMAND median_integration $<TARGET_FILE:median> --opencl)
  set_tests_properties(median_opencl median_opencl_integration PROPERTIES TIMEOUT 180 SKIP_RETURN_CODE 77)
endif()
