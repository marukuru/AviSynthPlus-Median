find_library(AVISYNTH_LIBRARY NAMES avisynth)
if(NOT AVISYNTH_LIBRARY)
  message(FATAL_ERROR "MEDIAN_BUILD_TESTS requires the AviSynth+ runtime library")
endif()
add_executable(median_integration integration.cpp)
set_target_properties(median_integration PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(median_integration PRIVATE ../Median)
target_link_libraries(median_integration PRIVATE ${AVISYNTH_LIBRARY})
add_test(NAME median_integration COMMAND median_integration $<TARGET_FILE:median>)
set_tests_properties(median_integration PROPERTIES TIMEOUT 120)
add_executable(median_kernels kernels.cpp ../Median/kernels.cpp)
target_include_directories(median_kernels PRIVATE ../Median)
add_test(NAME median_kernels COMMAND median_kernels)
